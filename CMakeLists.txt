cmake_minimum_required(VERSION 3.24)
project(FetchNetworkConfig VERSION 1.4.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 with MSVC is required")
endif()
enable_language(RC)
set(FNC_PUBLISHER "FetchNetworkConfig" CACHE STRING "Publisher name displayed in executable file properties")
configure_file(src/version.rc.in ${CMAKE_CURRENT_BINARY_DIR}/version.rc @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(miniz STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tdef.c
  third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(miniz PUBLIC third_party/miniz)
target_compile_definitions(miniz PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(fnc_core STATIC src/common.cpp src/worker.cpp src/collectors.cpp src/events.cpp src/report.cpp src/analyzer.cpp src/sdk_logs.cpp src/sapera.cpp src/mvsdk.cpp src/dumps.cpp)
target_include_directories(fnc_core PUBLIC src third_party/json)
target_compile_definitions(fnc_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 FNC_VERSION="${PROJECT_VERSION}")
target_compile_options(fnc_core PUBLIC /utf-8 /W4 /permissive- /EHsc /guard:cf)
target_link_libraries(fnc_core PUBLIC miniz iphlpapi ws2_32 wbemuuid ole32 oleaut32
  advapi32 setupapi wevtapi bcrypt xmllite shlwapi userenv)
add_executable(FetchNetworkConfig src/main.cpp ${CMAKE_CURRENT_BINARY_DIR}/version.rc)
target_link_libraries(FetchNetworkConfig PRIVATE fnc_core)
target_link_options(FetchNetworkConfig PRIVATE /MANIFESTUAC:level='asInvoker' /DYNAMICBASE /NXCOMPAT /guard:cf)
include(CTest)
if(BUILD_TESTING)
  add_executable(fnc_tests tests/tests.cpp)
  target_link_libraries(fnc_tests PRIVATE fnc_core)
  add_test(NAME fnc_behavior COMMAND fnc_tests)
  add_executable(fnc_analyzer_tests tests/analyzer_tests.cpp)
  target_link_libraries(fnc_analyzer_tests PRIVATE fnc_core)
  add_test(NAME fnc_analyzer COMMAND fnc_analyzer_tests)
  add_executable(fnc_benchmark_tests tests/benchmark_tests.cpp)
  target_link_libraries(fnc_benchmark_tests PRIVATE fnc_core)
  add_test(NAME fnc_benchmark COMMAND fnc_benchmark_tests)
  add_executable(fnc_mvsdk_tests tests/mvsdk_tests.cpp)
  target_link_libraries(fnc_mvsdk_tests PRIVATE fnc_core)
  add_test(NAME fnc_mvsdk COMMAND fnc_mvsdk_tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME fnc_deployment COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_deployment.py
      $<TARGET_FILE:FetchNetworkConfig> --output ${CMAKE_CURRENT_BINARY_DIR}/deployment-checks)
  endif()
endif()
install(TARGETS FetchNetworkConfig RUNTIME DESTINATION .)
install(FILES docs/CUSTOMER_GUIDE.md DESTINATION . RENAME README.md)
install(FILES docs/READ_ONLY_CONTRACT.md docs/RELEASE_NOTES.md DESTINATION docs)
install(FILES third_party/NOTICE.md DESTINATION licenses)
install(FILES third_party/json/LICENSE.MIT DESTINATION licenses RENAME json-LICENSE.MIT)
install(FILES third_party/miniz/LICENSE DESTINATION licenses RENAME miniz-LICENSE.txt)
